# GitHub Actions workflow — READY TO INSTALL BY HAND
#
# The connector used by Claude is not allowed to create files under .github/workflows (GitHub returns 403:
# pushing workflow files needs the `workflow` permission). To enable CI:
#   1. On GitHub: Add file > Create new file > name it  .github/workflows/ci.yml
#   2. Paste everything below the line "# ---- paste from here ----" and commit.
#   3. Open the Actions tab and read the first run. The `rules` job has NEVER been run: expect to fix it.
#
# ---- paste from here ----
name: CI

on:
  push:
  pull_request:

jobs:
  unit:
    name: Unit tests (Node ${{ matrix.node }})
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        node: [20, 22]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node }}
      # core/ and services/ use only node built-ins, so no npm install is needed.
      - run: npm test
        working-directory: functions

  rules:
    name: Firestore rules (Emulator)
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 21
      # NOTE: versions are not pinned yet (first CI run decides them; pin them afterwards).
      - run: npm install -g firebase-tools
      - run: npm install --no-save @firebase/rules-unit-testing firebase
        working-directory: functions
      - run: firebase emulators:exec --only firestore --project demo-qr-car "cd functions && node --test emulator-tests/rules.emulator.js"
