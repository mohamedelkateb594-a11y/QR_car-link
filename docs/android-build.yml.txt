# Ready-to-install CI for the Android APK.
# The GitHub integration used to push this repo cannot create files under .github/workflows (403), so it is parked here.
# To activate: GitHub > Add file > Create new file > name it  .github/workflows/android.yml  > paste this content > Commit.
# Then Actions > "Android build" > Run workflow. The APK appears under the run's Artifacts.
name: Android build

on:
  push:
    branches: ['android-app', 'android-app-complete']
    paths: ['android/**', '.github/workflows/android.yml']
  workflow_dispatch:

jobs:
  apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
      # The repo has no Gradle wrapper, so install a Gradle that matches AGP 8.5.2 (needs 8.7+).
      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'
      - name: Assemble debug APK
        working-directory: android
        run: gradle assembleDebug --no-daemon --stacktrace
      - uses: actions/upload-artifact@v4
        with:
          name: qr-car-debug-apk
          path: android/app/build/outputs/apk/debug/*.apk
          if-no-files-found: error
