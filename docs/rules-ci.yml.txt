# To activate: create .github/workflows/rules.yml on GitHub and paste this content (the integration cannot write workflows).
name: Database rules tests

on:
  push:
    branches: ['main']
    paths: ['database.rules.json', 'rules-tests/**', '.github/workflows/rules.yml']
  pull_request:
    paths: ['database.rules.json', 'rules-tests/**', '.github/workflows/rules.yml']
  workflow_dispatch:

jobs:
  rules:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '22'
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'
      - name: Run tests on the Realtime Database emulator
        working-directory: rules-tests
        run: |
          cp ../database.rules.json .
          npm install --no-audit --no-fund
          npm test
